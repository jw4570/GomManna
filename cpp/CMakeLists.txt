cmake_minimum_required(VERSION 3.24)
project(GomManna LANGUAGES CXX)
include(CTest)
option(GOMMANNA_BUILD_SERVER "Build HTTP server" ON)
option(GOMMANNA_USE_FAKE_REPOSITORY "Local fake data only" OFF)
option(GOMMANNA_TEST_WITHOUT_ROOM_SERVICE "Allow DB/API testing while the room module is unavailable" OFF)
option(GOMMANNA_BUILD_HTTP_DB_TESTS "Build opt-in HTTP tests against a real test database" OFF)
add_library(gm_contract INTERFACE)
target_include_directories(gm_contract INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(gm_contract INTERFACE cxx_std_20)
if(MSVC)
  target_compile_options(gm_contract INTERFACE /utf-8)
endif()
if(NOT GOMMANNA_BUILD_SERVER)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/geo/CMakeLists.txt")
    add_subdirectory(geo)
  endif()
  return()
endif()
include(FetchContent)
FetchContent_Declare(httplib
  GIT_REPOSITORY https://github.com/yhirose/cpp-httplib.git
  GIT_TAG v0.64.0 GIT_SHALLOW TRUE GIT_CONFIG core.longpaths=true)
FetchContent_Declare(json
  GIT_REPOSITORY https://github.com/nlohmann/json.git
  GIT_TAG v3.12.0 GIT_SHALLOW TRUE GIT_CONFIG core.longpaths=true)
FetchContent_MakeAvailable(httplib json)
find_package(OpenSSL 3 REQUIRED COMPONENTS Crypto)
add_executable(gommanna_server http/server.cpp)
target_link_libraries(gommanna_server PRIVATE gm_contract httplib::httplib nlohmann_json::nlohmann_json OpenSSL::Crypto)
if(GOMMANNA_USE_FAKE_REPOSITORY)
  target_compile_definitions(gommanna_server PRIVATE GOMMANNA_USE_FAKE_REPOSITORY=1)
  target_sources(gommanna_server PRIVATE http/fake_repository.cpp)
else()
  if(NOT TARGET gm_db)
    if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/db/CMakeLists.txt")
      add_subdirectory(db)
    else()
      message(FATAL_ERROR "Missing teammate gm_db implementation. Use -DGOMMANNA_USE_FAKE_REPOSITORY=ON for local development.")
    endif()
  endif()
  target_link_libraries(gommanna_server PRIVATE gm_db)
endif()
if(NOT GOMMANNA_USE_FAKE_REPOSITORY)
if(NOT TARGET gm_room)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/room/CMakeLists.txt")
    add_subdirectory(room)
  elseif(BUILD_TESTING AND GOMMANNA_TEST_WITHOUT_ROOM_SERVICE)
    # Only for HTTP/DB integration tests while the teammate's module is absent.
    # The placeholder returns 501 and never creates or manages real rooms.
    message(WARNING "Test-only mode: room creation, join, invite, leave and end are unavailable.")
    target_sources(gommanna_server PRIVATE tests/http/unavailable_room_service.cpp)
    target_compile_definitions(gommanna_server PRIVATE GOMMANNA_TEST_WITHOUT_ROOM_SERVICE=1)
  else()
    message(FATAL_ERROR "Missing teammate gm_room implementation. Do not replace real room service with a stub.")
  endif()
endif()
if(TARGET gm_room)
  target_link_libraries(gommanna_server PRIVATE gm_room)
endif()
endif()

if(BUILD_TESTING AND GOMMANNA_USE_FAKE_REPOSITORY)
  add_executable(gm_fake_smoke tests/http/fake_smoke.cpp http/fake_repository.cpp)
  target_link_libraries(gm_fake_smoke PRIVATE gm_contract OpenSSL::Crypto)
  add_test(NAME fake_repository_smoke COMMAND gm_fake_smoke)
endif()

if(BUILD_TESTING)
  add_executable(gm_http_environment_test tests/http/environment_test.cpp)
  target_link_libraries(gm_http_environment_test PRIVATE gm_contract)
  add_test(NAME http_environment COMMAND gm_http_environment_test)
endif()

if(BUILD_TESTING AND GOMMANNA_BUILD_HTTP_DB_TESTS AND NOT GOMMANNA_USE_FAKE_REPOSITORY)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_executable(gm_http_db_fixture tests/http/real_db_fixture.cpp)
  target_link_libraries(gm_http_db_fixture PRIVATE gm_contract gm_db)
  if(GOMMANNA_TEST_WITHOUT_ROOM_SERVICE AND NOT TARGET gm_room)
    set(GOMMANNA_TEST_ROOM_ARG --room-unavailable)
  endif()
  add_test(NAME http_real_db_integration
    COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/http/real_db_api_test.py
      --server $<TARGET_FILE:gommanna_server> --fixture $<TARGET_FILE:gm_http_db_fixture>
      ${GOMMANNA_TEST_ROOM_ARG})
  set_tests_properties(http_real_db_integration PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 180)
endif()
