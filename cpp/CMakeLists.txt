cmake_minimum_required(VERSION 3.24)
project(GomManna LANGUAGES C CXX)
include(CTest)
option(GOMMANNA_BUILD_SERVER "Build HTTP server" ON)
option(GOMMANNA_USE_FAKE_REPOSITORY "Local fake data only" OFF)
option(GOMMANNA_BUILD_HTTP_DB_TESTS "Build opt-in HTTP tests against a real test database" OFF)
add_library(gm_contract INTERFACE)
target_include_directories(gm_contract INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(gm_contract INTERFACE cxx_std_20)
if(MSVC)
  target_compile_options(gm_contract INTERFACE /utf-8)
endif()
if(NOT GOMMANNA_BUILD_SERVER)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/geo/CMakeLists.txt")
    add_subdirectory(geo)
  endif()
  return()
endif()
include(FetchContent)
FetchContent_Declare(httplib
  GIT_REPOSITORY https://github.com/yhirose/cpp-httplib.git
  GIT_TAG v0.64.0 GIT_SHALLOW TRUE GIT_CONFIG core.longpaths=true)
FetchContent_Declare(json
  GIT_REPOSITORY https://github.com/nlohmann/json.git
  GIT_TAG v3.12.0 GIT_SHALLOW TRUE GIT_CONFIG core.longpaths=true)
FetchContent_MakeAvailable(httplib json)
find_package(OpenSSL 3 REQUIRED COMPONENTS Crypto)
add_subdirectory(room)
add_subdirectory(geo)
add_library(gm_photo_image STATIC http/photo_image.cpp)
target_link_libraries(gm_photo_image PUBLIC gm_contract)
target_include_directories(gm_photo_image PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/third_party/stb)
add_executable(gommanna_server http/server.cpp http/photo_storage.cpp http/photo_service.cpp)
target_link_libraries(gommanna_server PRIVATE gm_contract gm_room gm_photo_image httplib::httplib nlohmann_json::nlohmann_json OpenSSL::Crypto)
if(GOMMANNA_USE_FAKE_REPOSITORY)
  target_compile_definitions(gommanna_server PRIVATE GOMMANNA_USE_FAKE_REPOSITORY=1)
  target_sources(gommanna_server PRIVATE http/fake_repository.cpp)
else()
  if(NOT TARGET gm_db)
    if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/db/CMakeLists.txt")
      add_subdirectory(db)
    else()
      message(FATAL_ERROR "Missing teammate gm_db implementation. Use -DGOMMANNA_USE_FAKE_REPOSITORY=ON for local development.")
    endif()
  endif()
  target_link_libraries(gommanna_server PRIVATE gm_db)
endif()
if(BUILD_TESTING AND GOMMANNA_USE_FAKE_REPOSITORY)
  add_executable(gm_fake_smoke tests/http/fake_smoke.cpp http/fake_repository.cpp)
  target_link_libraries(gm_fake_smoke PRIVATE gm_contract gm_room OpenSSL::Crypto)
  add_test(NAME fake_repository_smoke COMMAND gm_fake_smoke)
endif()

if(BUILD_TESTING)
  add_executable(gm_photo_image_test tests/http/photo_image_test.cpp)
  target_include_directories(gm_photo_image_test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/third_party/stb)
  target_link_libraries(gm_photo_image_test PRIVATE gm_photo_image)
  add_test(NAME photo_image COMMAND gm_photo_image_test)
  add_executable(gm_photo_service_test tests/http/photo_service_test.cpp http/photo_service.cpp http/fake_repository.cpp)
  target_include_directories(gm_photo_service_test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/http ${CMAKE_CURRENT_SOURCE_DIR}/third_party/stb)
  target_link_libraries(gm_photo_service_test PRIVATE gm_contract gm_room gm_photo_image OpenSSL::Crypto)
  add_test(NAME photo_service COMMAND gm_photo_service_test)
  add_executable(gm_room_service_test tests/room/room_service_test.cpp http/fake_repository.cpp)
  target_link_libraries(gm_room_service_test PRIVATE gm_contract gm_room)
  add_test(NAME room_service COMMAND gm_room_service_test)
  add_executable(gm_http_environment_test tests/http/environment_test.cpp)
  target_link_libraries(gm_http_environment_test PRIVATE gm_contract)
  add_test(NAME http_environment COMMAND gm_http_environment_test)
endif()

if(BUILD_TESTING AND GOMMANNA_BUILD_HTTP_DB_TESTS AND NOT GOMMANNA_USE_FAKE_REPOSITORY)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_executable(gm_http_db_fixture tests/http/real_db_fixture.cpp)
  target_link_libraries(gm_http_db_fixture PRIVATE gm_contract gm_db)
  add_executable(gm_geo_cli tests/geo/geo_cli.cpp)
  target_link_libraries(gm_geo_cli PRIVATE gm_geo)
  add_test(NAME http_real_db_integration
    COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/http/real_db_api_test.py
      --server $<TARGET_FILE:gommanna_server> --fixture $<TARGET_FILE:gm_http_db_fixture>
      --geo $<TARGET_FILE:gm_geo_cli>)
  set_tests_properties(http_real_db_integration PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 180)
  add_executable(gm_photo_db_test tests/db/photo_db_test.cpp http/photo_service.cpp)
  target_include_directories(gm_photo_db_test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/http ${CMAKE_CURRENT_SOURCE_DIR}/third_party/stb)
  target_link_libraries(gm_photo_db_test PRIVATE gm_contract gm_db gm_room gm_photo_image OpenSSL::Crypto)
  add_test(NAME photo_real_db_integration COMMAND gm_photo_db_test)
  set_tests_properties(photo_real_db_integration PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 90)
endif()
